<!-- ─────────────────── QUICK START ─────────────────── -->
<img src="https://capsule-render.vercel.app/api?type=rect&color=gradient&customColorList=0,2,20,22,15&height=50&text=%E2%9A%A1%20Quick%20Start&fontSize=22&fontColor=fff&animation=fadeIn" width="100%"/>

<br/>

```bash
# 1. Clone the repository
git clone https://github.com/Kunal2456/Business-KPI-Automation-System.git
cd Business-KPI-Automation-System

# 2. Install dependencies
pnpm install

# 3. Configure environment
cp .env.example .env
# Add your Supabase URL and keys to .env

# 4. Start development server
pnpm dev

# App is running at localhost — open in browser!
```
